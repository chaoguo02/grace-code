      if (ev.type === "run_terminal") {
        clearWatchdog();
        const re = ev as WsRunTerminalEvent;
        const tid = re.turn_id || "";
        patchSession(sid, (prev) => {
          const isActive = prev.activeTurn?.turnId === tid;
          const live = isActive ? prev.activeTurn
            : prev.completedTurns.find((t) => t.turnId === tid);
          const sourceTurn = live || (isActive ? prev.activeTurn : undefined);

          if (re.status === "completed") {
            const blocks = [...(sourceTurn?.assistantResponse.blocks || [])];
            if (re.summary && !blocks.some((b) => b.type === "text")) {
              blocks.push({ type: "text", content: re.summary });
            }
            const canonical: StreamingTurn = sourceTurn
              ? {
                  ...sourceTurn,
                  turnId: tid || sourceTurn.turnId,
                  runId: re.run_id || sourceTurn.runId,
                  assistantResponse: { ...sourceTurn.assistantResponse, blocks, status: "completed" as const },
                  meta: { ...sourceTurn.meta, steps: re.steps_taken ?? sourceTurn.meta.steps, tokens: re.total_tokens ?? sourceTurn.meta.tokens, completedAt: Date.now() },
                }
              : {
                  localId: `turn_${sid}_${re.turn_index || 0}`,
                  turnId: tid, runId: re.run_id || "", clientRequestId: "",
                  id: `turn_${sid}_${re.turn_index || 0}`,
                  userMessage: { id: `mu_${tid}`, blocks: [] },
                  assistantResponse: { id: `ma_${tid}`, blocks: [{ type: "text", content: re.summary || "" }], status: "completed" as const },
                  meta: { steps: re.steps_taken ?? 0, tokens: re.total_tokens ?? 0, startedAt: Date.now(), completedAt: Date.now(), eventSeq: ev.sequence || 0, hasGap: false },
                };
            return {
              ...prev,
              isRunning: false,
              steps: re.steps_taken ?? prev.steps,
              tokens: re.total_tokens ?? prev.tokens,
              streamingThought: "",
              activeTurn: isActive ? null : prev.activeTurn,
              completedTurns: upsertByTurnId(prev.completedTurns, canonical),
            };
          }
          const errorTurn: StreamingTurn | null = sourceTurn
            ? {
                ...sourceTurn,
                turnId: tid || sourceTurn.turnId,
                runId: re.run_id || sourceTurn.runId,
                assistantResponse: { ...sourceTurn.assistantResponse, status: "error" as const },
                meta: { ...sourceTurn.meta, completedAt: Date.now() },
              }
            : null;
          return {
            ...prev,
            isRunning: false,
            error: re.error || (re.status === "cancelled" ? "Run cancelled" : "Run failed"),
            planApproval: null,
            streamingThought: "",
            activeTurn: isActive ? null : prev.activeTurn,
            completedTurns: errorTurn ? upsertByTurnId(prev.completedTurns, errorTurn) : prev.completedTurns,
          };
        });
        void get().loadTimeline(sid, undefined, 0, tid);
        return;
      }
